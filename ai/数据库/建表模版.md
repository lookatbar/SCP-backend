`{表名}_id` CHAR(36) NOT NULL COMMENT '主键ID',
`{字段1}` VARCHAR(50) NOT NULL COMMENT '字段1说明（如：user_name 用户名）',
`{字段2}` DECIMAL(10,2) NOT NULL DEFAULT 0.00 COMMENT '字段2说明（如：amount 金额）',
`{字段3}` INT UNSIGNED NOT NULL DEFAULT 0 COMMENT '字段3说明（如：status 状态）',
`is_{布尔字段}` TINYINT UNSIGNED NOT NULL DEFAULT 0 COMMENT '布尔字段说明（1=是，0=否）',
`created_by` CHAR(36) NOT NULL DEFAULT '' COMMENT '创建者',
`created_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
`modified_by` CHAR(36) NOT NULL DEFAULT '' COMMENT '修改者',
`modified_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '修改时间',
`updated_time` DATETIME(3) NOT NULL DEFAULT CURRENT_TIMESTAMP(3) ON UPDATE CURRENT_TIMESTAMP(3) COMMENT '更改时间',

-- 主键索引（必选）
PRIMARY KEY (`{表名}_id`) COMMENT '主键索引',
-- 唯一索引（按需添加，业务唯一字段必加）
UNIQUE KEY `uk_{唯一字段}` (`{唯一字段}`) COMMENT '唯一索引：{唯一字段说明}',
-- 普通索引（按需添加，查询字段必加）
KEY `idx_{普通字段}` (`{普通字段}`) COMMENT '普通索引：{普通字段说明}',
-- 组合索引（按需添加，遵循最左匹配）
KEY `idx_{字段A}_{字段B}` (`{字段A}`, `{字段B}`) COMMENT '组合索引：{字段A}+{字段B}查询'
) ENGINE = InnoDB DEFAULT CHARSET = utf8mb4 COLLATE = utf8mb4_unicode_ci COMMENT = '{表说明}';
